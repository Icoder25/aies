from queue import PriorityQueue

def read():
    return [list(map(int, input().split())) for i in range(3)]


def h(s, goal):
    count = 0
    for i in range(3):
        for j in range(3):
            if s[i][j] != 0 and s[i][j] != goal[i][j]:
                count += 1
    return count


def astar(start, goal):
    pq = PriorityQueue()
    pq.put((h(start, goal), 0, start))
    visited = []

    while not pq.empty():
        f, g, s = pq.get()

        if s in visited:
            continue
        visited.append(s)

        print("Visited State:")
        for row in s:
            print(row)
        print("f =", f, "g =", g, "h =", h(s, goal))
        print()

        if s == goal:
            print("Goal Reached!")
            return

        x, y = 0, 0
        for i in range(3):
            for j in range(3):
                if s[i][j] == 0:
                    x, y = i, j

        moves = [(-1, 0), (1, 0), (0, -1), (0, 1)]

        for dx, dy in moves:
            nx = x + dx
            ny = y + dy

            if 0 <= nx < 3 and 0 <= ny < 3:
                new = [row[:] for row in s]

                new[x][y], new[nx][ny] = new[nx][ny], new[x][y]

                if new not in visited:
                    ng = g + 1
                    nh = h(new, goal)
                    pq.put((ng + nh, ng, new))

    print("No Solution Found")


print("Enter Initial State (0 = Blank):")
start = read()

print("Enter Goal State:")
goal = read()

astar(start, goal)

def cost(path, graph):
    total = 0
    n = len(path)

    for i in range(n):
        total += graph[path[i]][path[(i + 1) % n]]

    return total


def hill_climbing(graph):
    n = len(graph)
    path = list(range(n))

    while True:
        best = path[:]

        for i in range(1, n):
            for j in range(i + 1, n):
                new = path[:]
                new[i], new[j] = new[j], new[i]

                if cost(new, graph) < cost(best, graph):
                    best = new

        if best == path:
            break

        path = best

    print("Best Path:", path + [path[0]])
    print("Minimum Cost:", cost(path, graph))


n = int(input("Enter number of cities: "))
graph = []
print("Enter Cost Matrix:")
for i in range(n):
    graph.append(list(map(int, input().split())))

hill_climbing(graph)

def win(p):
    lines = [(0,1,2), (3,4,5), (6,7,8),
             (0,3,6), (1,4,7), (2,5,8),
             (0,4,8), (2,4,6)]

    for a, c, d in lines:
        if b[a] == b[c] == b[d] == p:
            return True
    return False

def minimax(p):
    if win("O"): return 1
    if win("X"): return -1
    if " " not in b: return 0

    best = -2 if p == "O" else 2

    for i in range(9):
        if b[i] == " ":
            b[i] = p

            next_p = "X" if p == "O" else "O"
            score = minimax(next_p)

            b[i] = " "

            if p == "O":
                best = max(best, score)
            else:
                best = min(best, score)

    return best

import nltk
from newspaper import Article
from nltk.tokenize import sent_tokenize, word_tokenize
from nltk.corpus import stopwords

nltk.download('punkt')
nltk.download('punkt_tab')
nltk.download('stopwords')

# Get article
article = Article('https://en.wikipedia.org/wiki/Influenza')
article.download()
article.parse()

sentences = sent_tokenize(article.text)
stop_words = set(stopwords.words('english'))

# Find answer
def get_answer(question):
    q = set(word_tokenize(question.lower())) - stop_words
    best, score = "", 0

    for s in sentences:
        words = set(word_tokenize(s.lower())) - stop_words
        count = len(q & words)

        if count > score:
            best, score = s, count

    return best

    else:
        answer = get_answer(user)
